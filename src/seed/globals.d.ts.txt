// The globals a snippet gets inside the sandbox (see PRELUDE in sandbox.ts), besides `webflow` from types/webflow.ts.
// Snippets are typechecked with these and the ES2023 library only, so Node and Bun APIs fail the typecheck.
export {};

declare global {
  // Values print as text; objects and arrays as compact JSON. All levels go to the same output.
  const console: {
    log(...values: unknown[]): void;
    info(...values: unknown[]): void;
    warn(...values: unknown[]): void;
    error(...values: unknown[]): void;
    debug(...values: unknown[]): void;
  };
  // Arguments after the snippet path: `wf-mcp run ~/.wf-mcp/snippets/x.ts a b` gives ["a", "b"].
  const args: readonly string[];
  // Files in out/, the only folder a snippet can reach. Paths are relative to out/.
  const out: {
    // Writes text, or other values as pretty JSON, and returns the file's full path.
    write(name: string, data: unknown): string;
    read(name: string): string;
    // MD5 of a file's bytes as lowercase hex: the `file_hash` that create_asset, create_font and replace_font_file take.
    md5(name: string): string;
    // Uploads a file's bytes to finish create_asset, create_font or replace_font_file. Pass that call's result
    // (or its { uploadUrl, uploadDetails }); only targets returned earlier in this run are accepted, each once.
    upload(name: string, target: unknown): Promise<{ status: number; bytes: number }>;
  };
  function sleep(ms: number): Promise<void>;

  // A CMS item as list_collection_items returns it. F is its fieldData: declare the field slugs the snippet reads
  // (get_collection_details lists them).
  type CmsItem<F = Record<string, unknown>> = {
    id: string;
    // Present when the site has CMS locales; with several, each locale's variant is a separate item with the same id.
    cmsLocaleId?: string;
    createdOn: string;
    lastUpdated: string;
    // When the item was last published, or null if it isn't on the live site.
    lastPublished: string | null;
    isDraft: boolean;
    isArchived: boolean;
    fieldData: F;
  };
  // Where an item stands against the live site. "On the site" is read from lastPublished: the MCP tools can't list
  // live items directly.
  //   archived          not on the site
  //   draft             never published; isDraft
  //   staged            never published; goes live with the next publish
  //   published         on the site, no unpublished edits
  //   changes-in-draft  on the site; edits saved as a draft (isDraft is true, yet the published version is live)
  //   changes-staged    on the site; edits go live with the next publish
  type CmsItemState = "archived" | "draft" | "staged" | "published" | "changes-in-draft" | "changes-staged";
  const cms: {
    // Every item list_collection_items returns for this input, all pages read (Webflow filters and sorts as usual).
    // Throws unless every page reports the same pagination.total and it got that many distinct items, which catches
    // most edits made while it reads; Webflow has no snapshot read, so it can't catch them all.
    listAll<F = Record<string, unknown>>(
      input: Omit<Parameters<typeof webflow.data_cms_tool.list_collection_items>[0], "request"> & {
        request?: Omit<NonNullable<Parameters<typeof webflow.data_cms_tool.list_collection_items>[0]["request"]>, "limit" | "offset">;
      },
    ): Promise<CmsItem<F>[]>;
    itemState(item: CmsItem<unknown>): CmsItemState;
    // On the live site: published, changes-in-draft or changes-staged. A draft that was published is still live.
    // Says only whether the item is on the site: its fieldData is the latest saved version, unpublished edits included.
    isLive(item: CmsItem<unknown>): boolean;
  };
}
