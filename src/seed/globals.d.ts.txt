// The globals a snippet gets inside the sandbox (see PRELUDE in sandbox.ts), besides `webflow` from types/webflow.ts.
// Snippets are typechecked with these and the ES2023 library only, so Node and Bun APIs fail the typecheck.
export {};

declare global {
  // Values print as text; objects and arrays as compact JSON. All levels go to the same output.
  const console: {
    log(...values: unknown[]): void;
    info(...values: unknown[]): void;
    warn(...values: unknown[]): void;
    error(...values: unknown[]): void;
    debug(...values: unknown[]): void;
  };
  // Arguments after the snippet path: `wf-mcp run ~/.wf-mcp/snippets/x.ts a b` gives ["a", "b"].
  const args: readonly string[];
  // Files in out/, the only folder a snippet can reach. Paths are relative to out/.
  const out: {
    // Writes text, or other values as pretty JSON, and returns the file's full path.
    write(name: string, data: unknown): string;
    read(name: string): string;
    // MD5 of a file's bytes as lowercase hex: the `file_hash` that create_asset, create_font and replace_font_file take.
    md5(name: string): string;
    // Uploads a file's bytes to finish create_asset, create_font or replace_font_file. Pass that call's result
    // (or its { uploadUrl, uploadDetails }); only targets returned earlier in this run are accepted, each once.
    upload(name: string, target: unknown): Promise<{ status: number; bytes: number }>;
  };
  function sleep(ms: number): Promise<void>;
}
